\subsection*{Indicatori di Prestazione}

\[
    \begin{array}{lcr}
        \text{as} & x & 3
    \end{array}
\]
\begin{multicols}{2}
    \emphblue{Indice di turnover} \quad
    $ \text{Turnover} = \frac{\text{Fatturato}}{\text{Valore Medio delle Scorte}} $

    \emphblue{Indice di Giacenza Media} \quad $ I_{\text{GM}} = \frac{1}{\text{Turnover}} $

    \emphblue{Fatturato} \quad $ F = \text{Domanda} \cdot \text{Valore} $

    \emphblue{Quantità media del magazzino} \quad $ QM = \frac{\text{Dimensione Ordine}}{2} $

    \emphblue{Valore medio delle scorte} \quad $ VM = QM \cdot \text{Valore} $
\end{multicols}
